A.DATA STRUCTURES

struct swnd_struct {
    int swnd_size; --the current available window size (maximum number of unacked messages allowed)
    int unacked[10]; --max 10 messages that have been sent but not acked
}

struct rwnd_struct {
    int rwnd_size; --the amount of free space currently available in the receiver-side message buffer.
    int expected[10]; --tracks the expected in-order sequence numbers for incoming messages.
}

struct message {
    int type; --indicates packet type (0 for Data, 1 for ACK).
    int seq_no; --the sequence number of the current data packet.
    int ack_no; --the sequence number of the current data packet.
    int rwnd_size; --the sequence number of the current data packet.
    int msg_len; --the sequence number of the current data packet.
    char msg_data[512]; --the sequence number of the current data packet.
}


struct sock_info {
    int not_free; --flag indicating if the shared memory slot is in use (1) or available (0)
    pid_t ppid; --Process ID of the user application that created the socket (used for garbage collection)
    int fd_udp; --The underlying OS-level UDP file descriptor managed by the daemon
    char src_IP[16]; --The local IP address and port requested by the user application
    int src_port;    
    char IP[16]; --The remote destination IP address and port
    int port;        
    int bind_done; --State machine flag indicating bind status (0=Init, 1=Bind Requested, 2=Bound by Daemon, -1=Close Requested)
    int cur_seq_no; --the next sequence number to assign to a new outgoing message
    
    message send_buffer[SEND_BUF_SIZE];
    int send_buffer_sz; --Array storing messages queued by the application to be sent, and its current size
    
    int app_read_seq_no; --The specific sequence number the application's k_recvfrom call is waiting to read next
    message recv_buffer[RECV_BUF_SIZE]; --Array storing received in-order/out-of-order messages waiting to be read, and its current size
    int recv_buffer_sz;

    swnd_struct swnd; --Instances of the sender and receiver window tracking structures
    rwnd_struct rwnd;
    long long send_times[10]; --Timestamps (in milliseconds) tracking when each unacked message was last transmitted, used to trigger retransmissions
    int nospace; --Flag indicating if the receiver buffer has hit its maximum capacity (rwnd = 0)
    int total_transmissions; --Counter tracking the absolute number of packets sent (including retransmissions) for testing metrics
    int total_messages; --Counter tracking the unique number of messages generated by the application
    pthread_mutex_t mutex; --A process-shared mutex ensuring thread-safe read/write access to this specific socket entry
}


FUNCTIONS

KSOCKET.H/KSOCKET.C

a.k_socket(int domain, int type, int protocol)
--Attaches the calling process to the shared memory segment, finds a free sock_info slot, initializes its variables and mutex, 
and returns the index as the KTP socket descriptor. Returns -1 if no space is available.

b.k_bind(int sock_KTP, char* src_IP, int src_port, char* dest_IP, int dest_port)
--Writes the requested local and remote IP/Port bindings to the shared memory and signals the daemon to execute the underlying UDP bind.
Blocks until the daemon confirms the bind is complete.


c.k_sendto(int sock_KTP, const void* buf, size_t size, int flags, struct sockaddr *dest_addr, socklen_t addrlen)
--Verifies destination credentials, encapsulates the user's data payload into a message struct with the correct sequence number, 
and queues it in the send_buffer for the daemon's sender thread to transmit.

d.k_recvfrom(int sock_KTP, void* buf, size_t size, int flags, const struct sockaddr *dest_addr, socklen_t *addrlen)
--Checks the recv_buffer for a message matching the application's expected sequence number. If found, copies the payload to the user's buffer, 
removes the message, updates the receiver window size, and increments the expected sequence number.

e.k_close(int sock_ktp)
--Blocks the application until all queued messages and unacknowledged packets are successfully transmitted and ACKed. 
Then signals the daemon to close the OS socket and marks the shared memory slot as free.

f.drop_message(float p)
--Simulates an unreliable network channel by generating a random number. Returns 1 (drop packet) with probability $p$, and 0 (keep packet) otherwise. 
Called by the receiver thread before processing any incoming packet.

INITKSOCKET.C 

--main()
-Creates or attaches to a System V shared memory segment (shmget/shmat) 
    capable of holding an array of N sock_info structures.
-Initializes the array, marking all slots as free, and initializes the process-shared pthread_mutex for each slot 
    to ensure thread-safe access between the daemon and user applications.
-Spawns the R, S, and G threads and waits for their execution (acting as an infinite loop server).

--get_current_time_ms()
-A utility function utilizing gettimeofday to return the current system time in milliseconds, which is required for precise 
    timeout calculations in the S thread.

--garbage_collector()
-A cleanup function called upon daemon termination to detach (shmdt) and explicitly destroy (shmctl) the shared memory segment, 
    preventing memory leaks in the OS.

--R_func()
Role: Acts as the listener for all active KTP sockets.
Socket Management: Continuously scans the shared memory array. If a user application has requested a bind (bind_done == 1), the R thread creates the actual OS-level UDP socket (socket()), 
    binds it to the requested port, and updates the flag to 2.
Multiplexing: Uses select() with a 500ms timeout to monitor all bound UDP file descriptors simultaneously without blocking indefinitely.
Data Reception: When a packet arrives, it first checks drop_message(DROP_PROB) to simulate network unreliability. If the packet is kept, it processes it based on its type:
Type 0 (Data): Handles duplicate checking, stores in-order and out-of-order packets in the recv_buffer, updates the receiver window (rwnd), 
    and transmits an Acknowledgement (ACK) packet back to the sender.
Type 1 (ACK): Updates the sender window size (swnd), clears acknowledged sequence numbers from the unacked array, 
    and removes the successfully delivered packets from the send_buffer.
Window Probing: Handles edge cases where the receiver buffer is full (nospace == 1). Upon a select() timeout, if the user application has read data 
    and freed up space, it sends a duplicate ACK with the updated rwnd_size to unblock the sender


--S_func()
Role: Manages the transmission and retransmission of data packets queued by user applications.
Execution: Sleeps for T/2 seconds and wakes up periodically to evaluate the sender window state of all active KTP sockets.
Timeout Management: Compares the send_times of unacknowledged packets against the current time. 
    If the difference exceeds the timeout value T, it triggers a retransmission of all unacknowledged packets within the current window and increments the total_transmissions counter.
New Transmissions: Checks if the sender window (swnd_size) has available slots and if there are pending, unsent packets in the send_buffer. It transmits these new packets via sendto(), 
    logs their sequence numbers in the unacked array, records their initial send_times, and increments total_transmissions.


--G_func()
Role: Prevents resource exhaustion by cleaning up orphaned sockets.
Execution: Wakes up every 2 seconds to scan the sock_info array.
Cleanup Logic: If a socket is marked as active (not_free == 1), it uses the kill(ppid, 0) system call to check if the user 
    process that created the socket is still alive. If the system call returns -1 (indicating the user process crashed or was forcefully killed before calling k_close()), 
    the G thread explicitly closes the underlying OS UDP descriptor and resets the shared memory slot, making it available for future connections.




USER1.C-sender application

Initialization: Parses command-line arguments for source/destination IPs and ports, along with the file to send. Defaults to binding at 127.0.0.1:8080 and sending to 127.0.0.2:9090.
Socket Setup: Calls k_socket() to initialize a KTP socket and k_bind() to register the addresses with the background daemon.
File Reading: Opens the target file in binary read mode (rb).
Transmission Loop: Reads the file in chunks of up to 512 bytes. It uses a retry loop with k_sendto(), sleeping for 10ms (usleep(10000)) if the sender buffer is temporarily full (ENOSPACE).
Termination: Once the file is fully read, it sends a final 0-byte packet to act as an End-of-File (EOF) signal to the receiver. It then safely calls k_close() to block until all data is acknowledged before exiting.




USER2.c-receiver application

Initialization: Parses command-line arguments, defaulting to binding at 127.0.0.2:9090 and listening for traffic from 127.0.0.1:8080.
Socket Setup: Creates the KTP socket and binds it to the specified listening ports.
File Writing: Opens or creates the target output file in binary write mode (wb).
Reception Loop: Enters a continuous loop calling k_recvfrom(). If no message is currently available in the buffer (ENOMESSAGE), it sleeps for 10ms to prevent high CPU utilization.
Data Processing: Upon receiving a data chunk, it writes the payload to the file and prints a progress dot (.) to the terminal.
Termination: If k_recvfrom() returns 0 bytes, it recognizes the EOF signal, breaks the reception loop, closes the file, and gracefully shuts down the socket using k_close().
